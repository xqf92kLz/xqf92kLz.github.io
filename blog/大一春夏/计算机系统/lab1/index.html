    MUX4T1¶

    Logisim 绘制 .circ¶

    打开 Logisim。

    • S (选择信号)：它的 Data Bits 是 2 位。

    • Splitter (分线器)：把 2 位的 S 分成了两根单线，一根为 \(S_1\) (高位)，一根为 \(S_0\) (低位)。

      选择 Design→Wiring→Splitter，接在 S 后面。Fan Out 表示输出多少个 Branch，Bit Width In 表示输入多少个 bit。

      在 Properties 可以看到 Bit 0（低位）被分配到了 Branch 0（上面那根线），Bit 1（高位）被分到了 Branch 1（下面那根线）。在图中的 Splitter 接口处也可以看到数字 0,1，数字 0 对应的线就是 \(S_0\)，数字 1 对应的线就是 \(S_1\)。

    • 然后再连出四根线，表示 \(S_0,\bar S_0,S_1,\bar S_1\)（需要两个 NOT Gate）。

    • 再连到四个 AND Gate，第一个 AND Gate 只在 \(S=00\)（即 \(\bar{S_1} \cdot \bar{S_0}\)）时输出 1，第二个 AND Gate 只在 \(S=01\)（即 \(\bar{S_1} \cdot S_0\)）时输出 1，以此类推。

      这部分其实是使能逻辑。

    • I0, I1, I2, I3 (数据信号)

      用四个 AND Gate 把左侧的“使能信号”和真实的数据 \(I_0 \sim I_3\) 相乘。

    • 最后用一个 OR Gate 汇总，输出到 O。

    • 保存为 Mux4T1_1.circ。并点击 Design→main 然后修改 Properties→Circuit Name 为 Mux4T1_1。

      最终

    可以点击菜单栏 Project → Analyze Circuit，进行验证。

    SegDecoder¶

    使用结构化描述或者数据流描述方式完善代码仓库中的 src/lab1-2/submit/SegDecoder.v 文件：

    `timescale 1ns / 1ps

module SegDecoder (
    input wire [3:0] data,  // 要显示的十六进制数
    input wire point,       // 是否显示小数点，1 显示小数点
    input wire LE,          // 是否使能，0 使能

    output wire a,          
    output wire b,
    output wire c,
    output wire d,
    output wire e,
    output wire f,
    output wire g,
    output wire p           // 小数点对应的数码管
);

    // 信号分解与取反
    wire wire0, wire1, wire2, wire3;
    wire inv_wire0, inv_wire1, inv_wire2, inv_wire3;
    assign {wire3, wire2, wire1, wire0} = data;
    assign {inv_wire3, inv_wire2, inv_wire1, inv_wire0} = {~wire3, ~wire2, ~wire1, ~wire0};

    // 第一级：产生 16 个最小项 (AND Gates)。例如 and_gate[1] 对应二进制 0001
    wire [15:0] and_gate;
    assign and_gate[0]  = inv_wire3 & inv_wire2 & inv_wire1 & inv_wire0; // 0
    assign and_gate[1]  = inv_wire3 & inv_wire2 & inv_wire1 & wire0;     // 1
    assign and_gate[2]  = inv_wire3 & inv_wire2 & wire1     & inv_wire0; // 2
    assign and_gate[3]  = inv_wire3 & inv_wire2 & wire1     & wire0;     // 3
    assign and_gate[4]  = inv_wire3 & wire2     & inv_wire1 & inv_wire0; // 4
    assign and_gate[5]  = inv_wire3 & wire2     & inv_wire1 & wire0;     // 5
    assign and_gate[6]  = inv_wire3 & wire2     & wire1     & inv_wire0; // 6
    assign and_gate[7]  = inv_wire3 & wire2     & wire1     & wire0;     // 7
    assign and_gate[8]  = wire3     & inv_wire2 & inv_wire1 & inv_wire0; // 8
    assign and_gate[9]  = wire3     & inv_wire2 & inv_wire1 & wire0;     // 9
    assign and_gate[10] = wire3     & inv_wire2 & wire1     & inv_wire0; // A
    assign and_gate[11] = wire3     & inv_wire2 & wire1     & wire0;     // B
    assign and_gate[12] = wire3     & wire2     & inv_wire1 & inv_wire0; // C
    assign and_gate[13] = wire3     & wire2     & inv_wire1 & wire0;     // D
    assign and_gate[14] = wire3     & wire2     & wire1     & inv_wire0; // E
    assign and_gate[15] = wire3     & wire2     & wire1     & wire0;     // F

    // 第二级：段码组合逻辑 (OR Gates)
    // 这里的逻辑是：找出哪些数字下该段是“灭”的（即输出1）
    // 具体来说，数码管的每一段（a 到 g）在显示不同的数字时，有的要亮，有的要灭。以 a 段（最上面那横）为例，在显示数字 0, 2, 3, 5, 6, 7, 8, 9, A, E, F 时，它都需要亮；数字 1, 4, B, C, D 时它是不亮的。在共阳极电路中，就是要显示不亮的数字时阴极接入 1。

    // a 段熄灭的数字: 1, 4, B, C, D
    assign a = and_gate[1] | and_gate[4] | and_gate[11] | and_gate[12] | and_gate[13] | LE; //LE = 1 时数码管必须全灭

    // b 段熄灭的数字: 5, 6, B, C, E, F
    assign b = and_gate[5] | and_gate[6] | and_gate[11] | and_gate[12] | and_gate[14] | and_gate[15] | LE;

    // c 段熄灭的数字: 2, C, E, F
    assign c = and_gate[2] | and_gate[12] | and_gate[14] | and_gate[15] | LE;

    // d 段熄灭的数字: 1, 4, 7, A, F
    assign d = and_gate[1] | and_gate[4] | and_gate[7] | and_gate[10] | and_gate[15] | LE;

    // e 段熄灭的数字: 1, 3, 4, 5, 7, 9
    assign e = and_gate[1] | and_gate[3] | and_gate[4] | and_gate[5] | and_gate[7] | and_gate[9] | LE;

    // f 段熄灭的数字: 1, 2, 3, 7, D
    assign f = and_gate[1] | and_gate[2] | and_gate[3] | and_gate[7] | and_gate[13] | LE;

    // g 段熄灭的数字: 0, 1, 7, C
    assign g = and_gate[0] | and_gate[1] | and_gate[7] | and_gate[12] | LE;

    // 小数点 p: point=1 时显示(输出 0)，point=0 时灭(输出 1)
    // 逻辑：p = ~point | LE
    assign p = (~point) | LE;

endmodule
